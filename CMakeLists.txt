cmake_minimum_required(VERSION 3.21)
project(sdrangel_dqpsk_demod VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(dqpskcore STATIC src/dqpskcore.cpp)
target_include_directories(dqpskcore PUBLIC src)
add_executable(dqpsk_core_cli tests/core_cli.cpp)
target_link_libraries(dqpsk_core_cli PRIVATE dqpskcore)
if(WIN32)
  add_executable(dqpsk_app_diagnostic tests/app_diagnostic.cpp)
  target_link_libraries(dqpsk_app_diagnostic PRIVATE dbghelp)
endif()
option(BUILD_SDRANGEL_PLUGIN "Build native SDRangel Windows DLL" ON)
if(BUILD_SDRANGEL_PLUGIN)
  if(NOT SDRANGEL_SOURCE_DIR OR NOT SDRANGEL_IMPORT_DIR)
    message(FATAL_ERROR "Set SDRANGEL_SOURCE_DIR to matching SDRangel sources and SDRANGEL_IMPORT_DIR to import libraries")
  endif()
  find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network)
  set(CMAKE_AUTOMOC ON)
  qt_add_plugin(demoddqpsk CLASS_NAME DQPSKPlugin)
  target_sources(demoddqpsk PRIVATE src/dqpskplugin.cpp src/dqpskplugin.h src/dqpskdemod.cpp src/dqpskdemod.h src/dqpskgui.cpp src/dqpskgui.h)
  target_compile_definitions(demoddqpsk PRIVATE SDR_RX_SAMPLE_24BIT USE_SSE2=1 WIN32 _USE_MATH_DEFINES)
  target_include_directories(demoddqpsk PRIVATE "${SDRANGEL_SOURCE_DIR}/sdrbase" "${SDRANGEL_SOURCE_DIR}/sdrgui" "${SDRANGEL_SOURCE_DIR}/exports")
  target_link_libraries(demoddqpsk PRIVATE dqpskcore Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network "${SDRANGEL_IMPORT_DIR}/sdrbase.lib" "${SDRANGEL_IMPORT_DIR}/sdrgui.lib")
  set_target_properties(demoddqpsk PROPERTIES PREFIX "")
  add_executable(dqpsk_load_test tests/load_test.cpp)
  target_include_directories(dqpsk_load_test PRIVATE "${SDRANGEL_SOURCE_DIR}/sdrbase" "${SDRANGEL_SOURCE_DIR}/exports")
  target_link_libraries(dqpsk_load_test PRIVATE Qt6::Core "${SDRANGEL_IMPORT_DIR}/sdrbase.lib")
  add_executable(dqpsk_channel_test tests/channel_test.cpp)
  target_compile_definitions(dqpsk_channel_test PRIVATE SDR_RX_SAMPLE_24BIT)
  target_include_directories(dqpsk_channel_test PRIVATE "${SDRANGEL_SOURCE_DIR}/sdrbase" "${SDRANGEL_SOURCE_DIR}/exports")
  target_link_libraries(dqpsk_channel_test PRIVATE Qt6::Core "${SDRANGEL_IMPORT_DIR}/sdrbase.lib")
  add_executable(dqpsk_gui_preview tests/gui_preview.cpp)
  target_compile_definitions(dqpsk_gui_preview PRIVATE SDR_RX_SAMPLE_24BIT)
  target_include_directories(dqpsk_gui_preview PRIVATE "${SDRANGEL_SOURCE_DIR}/sdrbase" "${SDRANGEL_SOURCE_DIR}/sdrgui" "${SDRANGEL_SOURCE_DIR}/exports")
  target_link_libraries(dqpsk_gui_preview PRIVATE Qt6::Widgets "${SDRANGEL_IMPORT_DIR}/sdrbase.lib" "${SDRANGEL_IMPORT_DIR}/sdrgui.lib")
endif()
if(MSVC)
  target_compile_options(dqpskcore PRIVATE /W4 /O2)
endif()
